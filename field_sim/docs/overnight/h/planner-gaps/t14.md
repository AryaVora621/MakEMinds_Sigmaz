## Proposal
Keep the scene-driven tip cells (L49-56) on top of the new hive. Replace L48 and L57-61.
Coordinates: `size` = half-width in inches (use 22.77 = 45.5/2 from scene.json). `HIVE_ASPECT` = 24.89/22.77 (half-height / half-width, from the same bounds). Honeycomb cell size (0.2 * size) and petal proportions are STYLE CHOICES, not measured. Colors are plain hex.

```js
const HIVE_ASPECT = 24.89 / 22.77;   // from scene.json cell bounds (half-h / half-w)
const HONEY = { fill: '#f0b429', cell: '#e09a12', gap: '#8a5a00', rim: '#4a2e00', petal: '#ffd54f', petalRim: '#b8860b', core: '#6b3f00' };
function hexPath(ctx, x, y, r) {
  ctx.beginPath();
  for (let i = 0; i < 6; i++) {
    const a = Math.PI / 6 + i * Math.PI / 3;          // pointy-top
    const px = x + r * Math.cos(a), py = y + r * Math.sin(a);
    if (i) ctx.lineTo(px, py); else ctx.moveTo(px, py);
  }
  ctx.closePath();
}
// cx, cy: canvas px centre; size: half-width in field inches; scale: canvas px per inch
export function drawHive(ctx, cx, cy, size, scale) {
  const hw = size * scale, hh = size * HIVE_ASPECT * scale, r = size * 0.2 * scale;
  const dx = Math.sqrt(3) * r, dy = 1.5 * r, nR = Math.ceil(hh / dy) + 1, nC = Math.ceil(hw / dx) + 2;
  ctx.save();
  ctx.beginPath(); ctx.rect(cx - hw, cy - hh, 2 * hw, 2 * hh); ctx.clip();
  ctx.fillStyle = HONEY.fill; ctx.fillRect(cx - hw, cy - hh, 2 * hw, 2 * hh);
  ctx.lineWidth = Math.max(1, r * 0.12); ctx.strokeStyle = HONEY.gap;
  for (let row = -nR; row <= nR; row++) {
    for (let col = -nC; col <= nC; col++) {
      hexPath(ctx, cx + col * dx + (row & 1 ? dx / 2 : 0), cy + row * dy, r);
      ctx.fillStyle = HONEY.cell; ctx.fill(); ctx.stroke();
    }
  }
  ctx.restore();
  ctx.lineWidth = Math.max(2, r * 0.35); ctx.strokeStyle = HONEY.rim;   // darker rim, drawn after the clip
  ctx.strokeRect(cx - hw, cy - hh, 2 * hw, 2 * hh);
}
// x, y: canvas px centre; r: canvas px radius of the whole flower
export function drawFlower(ctx, x, y, r) {
  const pr = r * 0.42, off = r * 0.58;
  ctx.save();
  ctx.fillStyle = HONEY.petal; ctx.strokeStyle = HONEY.petalRim; ctx.lineWidth = Math.max(1, r * 0.12);
  for (let i = 0; i < 6; i++) {
    const a = i * Math.PI / 3;
    ctx.beginPath(); ctx.arc(x + off * Math.cos(a), y + off * Math.sin(a), pr, 0, Math.PI * 2);
    ctx.fill(); ctx.stroke();
  }
  ctx.beginPath(); ctx.arc(x, y, r * 0.3, 0, Math.PI * 2); ctx.fillStyle = HONEY.core; ctx.fill();
  ctx.restore();
}
```
